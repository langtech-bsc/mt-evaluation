اَساݙی خبراں چِٹّھی کِیتے سائن اَپ کرو اَتے سارا کُجھ ساریاں کنوں پہلاں ڄاݨو! اَساݙے ہزاراں راضی تھئے ڳاہکاں دی بِیادری وِچ رَلّو تُہاݙی ذاتی ڄاݨکاری کِین٘ویں وَرتی وین٘دی ہِے اِین٘دی وادُھو تفصیل کِیتے، مہربانی کر تے اَساݙی راز داری پالیسی ݙیکھو۔
اَساں تُہاکوں سِدھ تے رہوݨ وِچ مَدَت کرین٘دے ہِسے:\n• ہِک سادہ سَوکھا اَتے خوشگوار انٹرفیس\n• ہِِک نِشابر، سوہݨاں اَتے آپݨی مرضی نال بݨن جوڳا ڈیش بورڈ\n• سارے عام سان٘جھے ٹولز نال ڳان٘ڈھا\n• کہیں وی مسلیاں کِیتے، 24/7 تکنیکی مَدَت
اَساݙے بارے وِچ\nاَساں نِکّے وَݙّے ہر کاروبار کوں آپݨیاں خِذمتاں پیش کرین٘دے ہِسے، بھان٘ویں جو ایہ سچ ہِے اَساݙے نال زیادہ تر وَݙّے کاروبار رابطہ کرین٘دے ہِن۔ اَساݙا کَم تُہاݙے کَم کوں سَوکھا بݨاوݨ ہِے۔ اَساݙی نِیّت چَنڳے معیار دیاں خِذمتاں ݙیوݨ ہِے تاں جو تُہاݙا عملہ آپݨیاں ساریاں کوشِتاں تُہاݙے مقصد دی کامیابی اُتے لا سڳّے۔ اَساݙے پَن٘دھ دا مُن٘ڈھ 10 ورھیاں کنوں پہلاں اِیں پَک نال ٻَدّھیا ڳِیا ہَئی جو اَساں کَم اَتے ذاتی حیاتی دے وِچالے بروبری وِڳڑݨ نال تِھیوݨ آلے نقصان دا حل پیش کر سڳدے ہِسے۔ اَساݙیاں ساریاں خِذمتاں اَتے مسئلیاں دے حَل ݙیکھو!
میݙی پڑھائی دے اَغاز وِچ، میکوں یاد ہِے جو میکوں ہِک سوال دا جواب ݙیوݨاں پِیا ہَئی: تُساں آپݨے ویلے دے ہِک گھنٹے دا کِتّی مالی مُل گِھن٘دی ہِیوے؟ اِیں مشق دا مقصد اَساکوں اِیں سچ دا احساس ݙیواوݨاں ہَئی جو اَساں دِل پشوری وِچ ٻہوں گَھٹ ویلا ونڄیسوں (اَتے اِیں سانڳے وَدھ کَم کریسوں) ڄیکر اَساں ایہ وِچار کروں جو دِل پشوری وِچ لَنگھایا ڳِیا ہِک گھنٹہ، کمائی آلے کَم دے ہِک ضائع تِھیوݨ آلے گھنٹے سَن٘وّاں ہِے۔ میں سچّی مُچّی سُچین٘دی ہاں جو اوہ نقطۂ نظر اُوں ویلے تے کِین٘ویں لاڳو تھی سڳدا ہِے ڄہڑا اَساں ہر ݙیہاڑے آپݨے گھر توں کَم دی جاء تے آوݨ ون٘ڄݨ وِچ لَنگھین٘دے ہِسے۔
طاگت یا بھوڳݨ دی سَکَت، راء وَکّھو وَکھ لڳدی ہِے، تاں اَساں اِین٘دے بارے کیا سمجھوں؟ گَھٹ کھاوݨ یا چَنڳا کھاوݨ اَتے جسمانی ورزش ڈھیر کرݨ، عام طور تے اَساکوں اِیہو آکھیا وین٘دا ہِے۔ پَر اَساکوں ایہ وی پتہ ہووݨاں چاہِیدا ہِے جو ایہ ورزش کیہڑی ونکی دی ہووے! ایہ ڳالھ ݙیکھݨ آلی ہِے جو ݙُو دھڑے بݨ ڳئے ہِن، ہِک بھوڳݨ دی سَکَت دے حَق وِچ اَتے ݙُوجھا طاگت دے حَق وِچ۔ اِین٘دے توں وَدھ، ایہ ݙُو ݙاں نہ صرف ہِک ݙُوجھے دے مُن٘ڈھوں اَپُٹ تھی سڳدے ہِن، بلکہ ہِک ݙُوجھے دے ویری وی تھی سڳدے ہِن۔
اَڄ کَل اَساکوں اکثر آکھیا وین٘دا ہِے جو اَساکوں حال وِچ جِیوݨ سِکّھݨاں چاہِیدا ہِے۔ بَھلا اِتّھاں کُئی اِیہجا بُھلیکھا لڳّݨ دا ݙَر کائنی ڄہڑا مستقبل کوں برباد کر سڳدا ہِے؟ لڳدا ہِے جو لوکاں کوں اَڄوکے پَل نال سُن٘ڄاݨ کراوݨ دا حون٘صلہ وَدھارا، اَتے اُنّھاں کوں صرف چَسّاں چاوݨ دی مَت کوں رَلا ݙیوݨاں سَوکھا ہِے۔ اَتے بھان٘ویں جو یولو تئیں ڳالھ نہ وی ون٘ڄے، ایہ تصور کِیتا ون٘ڄ سڳین٘دا ہِے جو ہر پَل کوں مَنّݨ وِچ اُتّی ای ویلا لائیا ون٘ڄے جو مستقبل دا سوچݨ کِیتے کُجھ نہ بَچّے۔
میں خاص طور تے اُنّھاں ساریاں ٹولیاں دا تھورا مَنّݨاں چاہن٘دا ہاں جِنّھاں دے پورھئے دی وجہ توں اَڄ اِتّھاں اَساݙا میل تھی سڳیا ہِے۔ ساریاں کنوں پہلاں، میں تُہاکوں ݙَساوݨاں چاہن٘داں جو مستقبل، منصوبیاں اَتے بَدلی بارے ڳالھ مُہاڑ کرݨ کِیتے اِیں جاء تے وَلدا آ تے میکوں ݙاڈھی خُشی تھئی ہِے۔ مستقبل اُوہو ہِے جِیکوں اَساں اُنّھاں تَکڑیاں نِیتاں نال رَل تے ٻَدّھݨ دا فین٘صلہ کریسوں ڄہڑے جِدّت دی مَدَت کریسِن۔ اِیں مقصد کِیتے، میں دَم لائے مُتُّوں کَم کرݨ دا عہد کرین٘دا ہاں تاں جو تُہاݙے اَلاء سُݨے ون٘ڄِن، اَساݙے طریقیاں کوں سَوکھا کِیتا ون٘ڄے، اَتے اوہ اَڑان٘داں ڄہڑیاں اَساݙی کامیابی دے راہ وِچ آن٘دیاں ہِن، اُنّھاں کوں ہَٹایا ون٘ڄے۔

سئیں، میکوں کَل اَدّھے ݙِین٘ہ دی چُھٹّی چاہِیدی ہِے۔\nکیا تِھیا، خیر تاں ہِے؟\nمیݙی بھیݨ دا ٻال ڄَمّیا ہِے، میں اُوکوں ݙیکھݨ ون٘ڄݨاں ہِے۔\nواہ واہ! ٻہوں ٻہوں مُبارخاں!
آنند، اَڄ تُہاݙی طبیعت کِین٘ویں ہِے؟\nبَھلا تُساں کَل شویتا نال ڈاکٹر کول ڳئے ہاوے؟\nجِیہا نیہا، میں کَل آنند کوں ڈاکٹر کول گِھن ڳئی ہَم۔\nلڳدا ہِِے جو آنند اَرام کرین٘دا پِیا ہِے۔\nسلام ٻیلیو، میں چَنڳا ہاں، ڈاکٹر دی دوا نے کمال کر ݙِتّا ہِے!
اَمّاں، میݙا سکول سَیل کِیتے وین٘دا پِیا ہِے اَتے میں ون٘ڄݨاں چاہن٘دا ہاں؟\nمیکوں کُئی مسئلہ کائنی، پَر تیکوں وݙیرے کنوں وی پُچھݨاں پوسی!\nنہ! اَبّا سئیں نہ کر ݙیسن۔\nتُوں ڄیکر پُچھسیں تاں پتہ لڳسی!\nچَنڳا، میں اُنّھاں کنوں پُچھداں۔
اوہو ۔۔۔ میکوں نِن٘در نِھیں پئی اَمدی!\nچُھٹکی، اَدّھی راتیں اِیں پاٹی ہوئی اَواز وِچ ڳاݨیں کیوں پئی ڳان٘دی ہیں؟\nہائے اَدی، میں سَم نِھیں سڳدا۔\nاوہ! تیکوں نِن٘در نِھیں آن٘دی اَتے تُوں سارے ٹَٻَّر کوں جَڳائی کھڑی ہیں!\nاَڑی مِٹھڑی، نِن٘در دا بھیت تاں ݙَس۔\nتَتّا کِھیر پِی اَتے اَکّھیں بند کر تے لَمّی تھی ون٘ڄ!
اَڄ فراگی دی ڈاکٹر نال ملاقات ہِے۔ فراگی اَتے اُون٘دی اَمّاں ڈاکٹر کول وین٘دے پئے ہِن۔ ڈاکٹر دا ناں سئیں روہِت ہِے اَتے اوہ فراگی دا ساک دار وی ہِے۔ فراگی دی اَمّاں اُوکوں آہدی ہِے جو ݙَن٘د مان٘ڄݨ اَتے سُتھرا کَچّھا پاوݨ نہ وِساریں کیوں جو ایہ اُون٘دی معمول دی جاچ ہِے۔ پَر فراگی کَچّھا پاوݨ بُھل وین٘دا ہِے اِیں کِیتے اُون٘دی اَمّاں اُون٘دے کِیتے کاغذ دا کَچّھا بݨین٘دی ہِے۔ سئیں روہِت نے فراگی دی جاچ کِیتی اَتے آکھیُس، فراگی، توں تاں ݙاڈھا تَکڑا ہیں۔
اوہ ہَولے ہَولے نیڑے آن٘دی ہِے اَتے اُون٘دیاں اَکّھیں وِچ ہِک اَݨ اَلایا سُکھ ہِے۔ اِین٘ویں لڳدا ہِے جِین٘ویں ہر قدم تے ویلا رُک وین٘دا ہِے، اَتے ڄڳ دیاں ساریاں خُشیاں اُون٘دے ہاسے وِچ وَسدیاں ہِن۔ میں اُوکوں تَکدا ہاں، اَتے محسوس کرین٘دا ہاں جِین٘ویں اُون٘دے دِل دی دھڑکن میݙے نال ڄُڑی کھڑی ہِے۔ کُجھ وی اَلاوݨ کنوں پہلاں، میں ٻس اُون٘دیاں اَکّھیں وِچ ڳُم تھی وین٘دا ہاں، جِتّھاں ہر پَل صرف اَساں ݙُون٘ہیں ای ہون٘دے ہِسے۔
"اُوں نے میݙے وَل ݙِٹّھا اَتے کِھل پئی اَتے آکھیُس، ""تُوں اِین٘ویں لڳدا ہیں جِین٘ویں کہیں ڳݙان٘ہ نے سمارٹ فون گِھدّا ہووے!""" "میں کاوِڑ نال آکھیا، ""وَل کیا تِھیا، تُوں کُئی وݙّی شہزادی لڳدی پئی ہیں؟""" وَل اَساں ݙُون٘ہائیں دا ہاسا نِکّل ڳِیا "وَل اُوں آکھیا، ""ڄیکر کُئی ڳݙان٘ہ سمارٹ فون چَلا سڳدا ہِے، تاں بَھلا تُوں اِیکوں ٹھیک کرݨ وِچ میݙی مَدَت کریسیں؟""" "میں آپݨی کِھل ݙَکّݨ دی کوشِت کرین٘دے ہوئے آکھیا، ""ݙیکھ، میں ڳݙان٘ہ ہاں اَتے میں تیݙے نال ڳالھ کرݨ کنوں کݙان٘ہ نِھیں تَھکدا!"""
اَڄ میں آپݨے سنڳتیاں نال ہِک انڈو چائنیز رسٹورنٹ وِچ ڳِیا۔ ڄہڑیلے اَساں رسٹورنٹ دے اندر وَڑے، تاں سبھ توں پہلاں عملے نے اَساݙی ݙاڈھی آدر بھاء کِیتی اَتے اَساکوں پاݨی پُچّھیا۔ ایہ ہِک چَنڳی ڳالھ ہَئی، جِین٘دے نال بندے کوں ہمیش سُکھ مِلدا ہِے۔ ماحول وی ݙاڈھا سکون آلا ہَئی، جِین٘دے وِچ تھوڑی جئیں جدید رنگ دی رَلّت وی ہَئی، ڄہڑی رات دے کھاݨے کِیتے بہترین ہَئی۔ اَساں اُتّھاں کَئی شئیں کھادیاں جِنّھاں وِچ عام انڈو چائنیز ذائقیاں دی چَنڳی رَلّت ہَئی۔ اَساں ݙاڈھی چَس چاتی بھان٘ویں جو سروس ذری جِتّی ڈِھلّی ہَئی۔
میں لِنکڈ اِن اُتّے آپݨی پروفائل اَپ ڈیٹ کِیتی ہِے؟ ہُݨ ویلا ہِے جو تُساں آپݨے ڳُݨ، تجربے اَتے کامیابیاں کوں سلیقے نال پیش کرو۔ میݙا ہمیش مَنّݨ ہِے جو ڄݙݨ تُہاکوں آپݨے اُتّے پَک ہووے اَتے تُساں آپݨے تجربے ساجھے کرو، تاں دُنیاں وی اُوں پَک کوں محسوس کرین٘دی ہِے۔ آپݨی مشہوری ہِک فَن ہِے اَتے اِین٘دا مُل تَݙان٘ہہ پون٘دا ہِے ڄݙݨ ایہ ٹِھیک طریقے نال کِیتی ون٘ڄے!
ڄیکر تُساں آپݨاں سُݨَھپ نِکھارݨ چاہن٘دے ہِیوے، تاں ایہ لِپ گلاس اصلُوں تُہاݙے کِیتے ہِے۔ ایہ لِپ گلاس نہ صرف تُہاݙے ٻُلّھاں کوں کُولا اَتے نمی آلا رکھین٘دا ہِے، بلکہ اِین٘دا رنگ ہِک ایہڄیں خاص لَشک ݙین٘دا ہِے ڄہڑی تُہاݙے سُݨھَپ کوں ودھین٘دی ہِے۔ بھان٘ویں تُساں سادہ رُوپ چاہن٘دے ہِیوے یا کہیں کَٹھ وِچ فیشن ایبل ݙِسّݨ چاہن٘دے ہِیوے، ایہ لِپ گلاس تُہاݙے ٻُلّھاں کوں ہر موقعے کِیتے ٹَھہن٘دا بݨین٘دا ہِے۔ ایہ ٻہوں ویلے تئیں ٹِکدا ہِے۔ اَتے ہا، اِین٘دی وَدھیا پیکنگ وی اِیکوں ہِک خاص سوغات بݨین٘دی ہِے — ہِک سچّا تے شاندار احساس! اِیکوں ہِک واری اَزما تے ݙیکھو، اَتے آپ پَرکھو جو ایہ تُہاݙے ٻُلّھاں کوں کیݙا سوہݨاں بݨین٘دا ہِے۔
ایہ پُھل ایݙا سوہݨاں ہِے جو اِین٘ویں لڳدے جِین٘ویں جَنّت دی کُئی دیوی اِیں پُھل وِچ موجود ہووے۔\nایہ ہِک ایہڄاں نظارہ ہِے ڄہڑا اَساݙا ہَاں ٹَھرین٘دا ہِے۔\nمیݙی رُوح اِین٘دے وِچ وَس ڳئی ہِے، ایہ کیݙی کمال تصویر ہِے۔
"نمیتا نے انوپم دی پیشکش تے اصلُوں سچ آکھیا ہَئی جو اوہ نویں لوکاں دا ""نا جائز فَیدہ چین٘دا"" پِیا ہِے۔"\nپیوش اَتے رتیش ݙاڈھے نِماݨے، ادب کرݨ آلے اَتے کھرے ہِن اَتے او ہمیش لوکاں دی مَدَت کِیتے تیار رہن٘دے ہِن۔\nانوپم سچّی ݙاڈھا سوجھواݨ ہِے پَر میکوں نِھیں پتہ جو او اِیں سیزن وِچ کیا کرݨ دی کوشِت کرین٘دا پِیا ہِے۔\nانوپم اِیں سیزن وِچ ݙاڈھیاں پُٹّھیاں ڳالھیں کرین٘دا پِیا ہِے۔\nتِرِیجھی پیشکش: اُوکوں پَک نال رولیا ڳِیا ہَئی، اَتے پیوش ساریاں کنوں چَنڳی چوݨ ہون٘دی۔
اوہ یار، ایہ منظر ݙیکھ تے محسوس تِھیا جو ڄن٘واترے اُچّا اَلاوݨ آلے اَتے حرکتاں آلے بندے ہون٘دے ہِن۔\nاُوں نے ایہڄیں ٹور ݙِکھائی جِین٘ویں کُئی بالی ووڈ ہِیرو پہلاں پوز بݨاوے اَتے وَل ٹُر ون٘ڄے۔\n"میکوں تاں ݙاڈھی کِھل آئی ڄہڑیلے ڄن٘واترے آکھیا، ""انّھاں کَن٘دھاں وِچ تاکڑیاں کیوں کائنی؟"""\nاوہ یار، ڄن٘واترے دا ݙاڈھا کمال دَرشن ہَئی۔
گھر وِچ خطریاں دی جاچ کرو اَتے کہیں وی ڈَٹّھے ہوئے ملبے کنوں پَرے رہوو۔ گھر دے اَندر اَتے ٻاہر محفوظ جائیں کوں سُن٘ڄاݨو اَتے زخمی لوکاں کوں اُتّھاں پُچاوݨ وِچ مَدَت کرو۔ ریسکیو سروسز نال رابطہ کرݨ کِیتے ہنگامی رابطے دا منصوبہ بݨاوݨ دی کوشِت کرو۔ بِیادری دی مَدَت کرو اَتے انّھاں اَوکھاں دی ڳویڑ کرو جِنّھاں اُتے دھیان ݙیوݨ دی لوڑ ہِے۔
اِیں ویلے وِچ، ٹیکنالوجی دے محفوظ اَتے ذِمّے وارانہ ورتارے کِیتے چَنڳے قاعدے قانون دی لوڑ ہِے۔ آوݨ آلے ݙاہ ورھیاں وِچ، بھارت ٹیکنالوجی دی ہِک وَݙّی طاگت بݨ تے نِکھرسی۔ ایہڄیں ویلے، ڈیجیٹل عوامی انفراسٹرکچر ہِک پُل دا کَم کریسی اَتے اِیکوں اَڑان٘د نِھیں سمجھݨاں چاہِیدا۔ بھارت دُنیاں دے بَھلے کِیتے آپݨاں ڈیجیٹل عوامی انفراسٹرکچر ڄڳ نال ساجھا کرݨ کِیتے تیار ہِے۔ بھارت ساری انسانیت دے حقّاں دی راکھی اَتے دُنیاں دی خوشحالی کِیتے ترقی دے کَم کرین٘دا رَہسی۔
سڑک پار کرݨ کِیتے زیبرا کراسنگ وَرتو۔\nپٹرول پمپ تے موبائل فون ورتݨ منع ہِے۔\nنشے وِچ ڳَݙّی چلاوݨ جوکھوں دا کَم ہِے۔\nآر ٹی او وِچ آپݨی ڳَݙّی دا اندراج کراوݨ یاد رکّھو۔
اِیں ورھے میلے وِچ ٻہوں رَش ہَئی، خاص کر شام دے ویلے۔ دُکانداراں دے مُون٘ھاں تے کِھل ہَئی کیوں جو وِکری وی ٻہوں چَنڳی تَھئی۔ لوکاں نے پِین٘گھاں، چاٹ اَتے ڳُڑ آلی جلیبی دیاں ݙاڈھیاں چَسّاں چاتیاں۔ میلے وِچ رَش کوں ݙیہدے ہوئے، پُلس دے انتظام وی چَنڳے ہَن۔ رَش دی وجہ کنوں، شہر وِچ ٹریفک جام وی ٻہوں ہَئی۔
سوشل میڈیا نے ڳالھ مُہاڑ دا ڈَھن٘ڳ بَدل ݙِتّا ہِے۔ لوک ہُݨ ہِک ݙُوجھے نال پہلاں کنوں ݙاڈھا فوری رابطہ کر سڳدے ہِن۔ اِین٘دے نال نال، ایہ وی سَچ ہِے جو لوکاں وِچ بَھرَم اَتے کُوڑیاں خبراں کِھن٘ݙاوݨ دا ݙَر وَدھ ڳِیا ہِے۔ ایہ نہ صرف لوکاں دیاں سوچاں کوں متاثر کرین٘دا پِیا ہِے، بلکہ این٘دے نال وَسیب وِچ آپت دی پُھٹ اَتے نفرت وی وَدھدی پَئی ہِے، ڄین٘دی وجہ توں ساجھے فین٘صلے کرݨ اَوکھا تِھین٘دا وین٘دا ہِے۔
ڄیکر تُہاکوں سموسے کھاوݨے ہِن تاں اَساں ݙیسوں، اَتے ڄیکر بݨاوݨے ہِن تاں اَساں سَمان وی پُچا ݙیسوں کیوں جو اَساں ہر شے پُچین٘دے ہِسے!\nفون کوں وَل وَل نہ ݙیکھو کیوں جو تُہاݙی ڈلیوری صرف 10 مِنٹ وِچ پُڄ ویسی۔\nکرکٹ میچ ہووے اَتے گھر کھاوݨ کِیتے کُجھ نہ ہووے، اَساں پُچیسوں اَتے ایہ پَکّی ڳالھ ہِے!

جوڳیاں، سنتاں، سادھوواں، سادھویاں، کلپ واسیاں اَتے یاتریاں، ساریاں کوں مہا کُمبھ پریاگ راج 2025 وِچ سِر اکّھیں تے! ایہ تہوار سِجھ، چَن٘در اَتے مشتری دیاں تاریاں وِچ خاص چال اُتے منایا وین٘دا ہِے۔ پریاگ راج وِچ 2025 دا مہا کُمبھ میلہ 13 جنوری 2025 کنوں شروع تھیسی اَتے 26 فروری 2025 تئیں رہسی۔ میلے دے چِھی ہفتیاں دے دوران، 40 کروڑ توں وَدھ لوکاں دے اِین٘دے وِچ رَلّݨ دی اُمین٘د ہِے۔
ذہنی صِحت دی سان٘بھ کوں اکثر اَݨ ݙِٹّھا کر ݙِتّا وین٘دا ہِے۔ اِتّھاں ذہنی صحت دی سان٘بھ دے کُجھ سَوکھے طریقے ہِن جِنّھاں اُتّے تُساں عمل کر سڳدے ہِیوے۔ اِیں ڳالھ دا پَک کرو جو تُساں لوڑِین٘دا اَرام کرین٘دے ہِیوے۔ اِین٘دے کنوں اَن٘ج، تُساں آپݨے ویلھے ویلے وِچ دھیان وی لا سڳدے ہِیوے۔ نِکّیاں کامیابیاں دی خُشی مناؤ۔
اَساݙے کول تُہاݙے کِیتے وَکّھو وَکھ وَنکی دے ڈیپازٹ ہِن۔ پہلاں، اَساݙے پراڈکٹس اَتے آپݨی لوڑ کوں ڄاݨ گِھنّو۔ ٹرم ڈیپازٹ وِچ، تُہاݙے پیسے ہِک خاص مُد تئیں سان٘بھ تے رکّھے ویسن۔
پلیٹس ورزش دی ہِک ایہجی وَنکی ہِے جِین٘دے وِچ ہَولے ہَولے اَتے دھیان نال حرکتاں کِیتیاں وین٘دیاں ہِن۔ نویں سِکّھݨ آلیاں کِیتے، ایہ صلاح ݙِتّی وین٘دی ہِے جو پلیٹس دی اَوکھائی کوں ہَولے ہَولے وَدھایا ون٘ڄے۔ پلیٹس دی اَوکھائی دے کَئی درجے ہِن، سَوکھے کنوں گِھن تے اَوکھے تئیں۔ اِتّھاں پلیٹس دیاں کُجھ مُن٘ڈھلیاں حرکتاں ݙِتّیاں ڳئیاں ہِن ڄہڑیاں تُساں اَزما سڳدے ہِیوے۔ آپݨے جُثّے کوں یوگا میٹ اُتے لَمّا کرو اَتے ٻان٘ہواں پاسیاں تے رکّھو۔
جی اییُن ٹَک، میں مُدتاں کنوں تیݙی تانگھ وِچ ہَم۔\nکِم شِن، تُساں ایݙا چِر میݙی تانگھ کیوں کِیتی؟ بَھلا ایݙی مُد اُݙِیک تے تُساں کَلّھے نِھیوے تھی ڳئے؟
یار، تُوں کَل کیا مُکالا مَلین٘دا ریہا ہیں؟ میݙیاں کالاں ای نِھیں چاتیاں!\nہا یار، معذرت۔ کَل میݙا موبائل پَڳُّو وِچ ڈَھیہ پِیا ہَئی۔
سلام، میݙے سوہݨے شاگردو۔ بَھلا تُساں گھر دا کَم کر گِھدّا ہِے ڄہڑا میں کَل ݙِتّا ہَئی؟\nحالی نِھیں، سئیں، کیوں جو میں حالی تئیں لائبریری کنوں کتاباں دی اُݙِیک وِچ ہاں۔\nمیݙا وی اِیہو حال ہِے، میکوں وی حالی تئیں کتاب نِھیں لَبّھی۔\nڄیکر ڳالھ اِین٘ویں ہِے تاں میں تُہاکوں رعایت ݙیوݨاں چاہن٘دا ہاں، مُد خَمِیس تئیں اَڳّوں کر ݙِتّی ویسی۔
میں نِھیں ڄاݨدا جو ایہ ٻُڈھڑی تِرِیمت کوݨ ہِے؟\nمالِن، اِین٘ویں نہ کر، پُتر۔ میں تیݙی اَمّاں ہاں!\nبَھلا ایہ سچی مُچی تیݙی وݙیری ہِے، مالِن؟\nاَصلُوں کائناں!
ݙاݙی اَمّاں، تُہاکوں پَک ہِے جو اَساں اَبّے دی اجازت مُتُّوں شِملہ ون٘ڄ سڳدیاں ہِسے؟\nجِیہا، خَیر ہِے۔ تیݙا پِیؤ ٻہوں فِکر کرین٘دا ہِے۔\nاَساں کَل ڄُلدیاں پئیاں ہِسے، انجلی۔
لالا کوں ہر سویل سکول ون٘ڄݨ بھان٘دا ہِے۔ رستے تے، لالا اکثر جَنڳلی ٻِلّیاں کوں بَھن٘بھیریاں نال کھیݙدیاں ݙیہدی ہِے۔ اوہ وَستی جِتّھاں لولو رہن٘دی ہِے، ٻہوں نِکڑی اَتے سکون آلی ہِے۔
اُوں سکول وِچ، رنگا اَتے سنتا دے وِچالے ٻہوں ساریاں یاداں ڄُڑیاں ہوئیاں ہَن۔ میگزین بورڈ اُنّھاں دے پریم دے ودھارے دا مُن٘ڈھ ہَئی۔ ڈھاٻہ جِتّھاں اوہ اکثر جماعت دے بعد مِلدے ہَن۔ اوہ پَھٹّہ نہ وِسارو، جِین٘دے اُتّے اُنّھاں دے ناں کُھدے ہوئے ہَن۔
مُنڈھلے ویلے اُنّھاں دے 5 رُکن ہَن۔ پَر ٹومی چِٹّے رینجر دے طور تے رَل ڳِیا۔ اِیں سان٘گے، ہُݨ رینجرز دا دستہ 6 بندیاں دا ہِے۔
ایہ ہوٹل اِیں شہر وِچ میݙی بہترین لَبھ وِچّوں ہِک ہَئی۔ کمرہ کُھلّا، صاف سُتھرا، اَتے چَنڳی طرحاں سان٘بھیا ہوئیا ہَئی جِین٘دے وِچّوں میراپی پہاڑ دا سوہݨاں نظارہ ہَئی، سچی مُچی کمال۔ مناسب ڄاء ہووݨ ہِک وݙّا فَیدہ ہَئی۔ ایہ ہوٹل بھیڑ بھاڑ دے وِچالے ہِے، پَر کمرے دی اَواز روکݨ دی صلاحیت نے اِین٘دا سکون قائم رکّھیا۔ ڄیکر میں کݙان٘ہیں وَلا یوگیا ڳِیا تاں پَک نال اِیں ہوٹل وِچ وَلّݨ دی کوشِت کریساں!
میں بصری فنون اَتے میڈیا ڈیزائن وِچ آرڈر گِھنّݨ کِیتے تیار ہاں۔ انگریزی اَتے انڈونیشین ݙُون٘ہائیں وِچ رواں ڳالھ مُہاڑ۔ حالے صرف مُلکی آرڈر گِھن٘دا پِیا ہاں۔ ہِک ماہر دے طور تے میݙے کول ویب اَتے کمپنی لوگو ڈیزائن دے آرڈر پورے کرݨ دا 15 ورھیاں کنوں وَدھ دا تجربہ ہِے۔ ڈیزائن دی وَنکی گرافیکل یا ہَتھ نال بݨائی ون٘ڄ سڳین٘دی ہِے۔ میݙے کَم دے نمونیاں کِیتے، مہربانی کر تے سِدّھا سنیہڑا گَھل تے رابطہ کرو۔
بَھلا اِیں ویڈیو گیم کوں بݨاوݨ آلی حالی تئیں اوہا جاپانی ٹولی ہِے؟\nنہ، ہُݨ ایہ پولش ٹولی ہِے پَر ایہ حالے وی کمال ہِے\nاوہ میں سمجھیا میکوں ہُݨ ڄاݨن دا شون٘ق تِھیا ہِے، بَھلا تئیں ایہ کھیݙی ہِے؟\nحالے تاں مُن٘ڈھلے لیول ہِن پَر گرافکس دا کُئی ڄوڑ کائنی، تُساں ݙاڈھا سرمایہ لڳیا ہوئیا ݙیکھ سڳدے ہِیوے۔
ایشیا وِچ ماس مُتُّوں کھاݨاں لَبّھݨ ݙاڈھا اَوکھا ہِے کیوں جو تقریباً ہر شے وِچ جانوراں توں بݨیاں شئیں ہون٘دیاں ہِن۔\nسچی ڳالھ تاں ایہ ہِے جو نہ، ایشیا دے ٻہوں سارے کھاݨے صرف سبزیاں آلے ہون٘دے ہِن۔\nٻس ڳالھ ایہ ہِے جو، تُہاکوں پتہ ہووݨاں چاہِیدا ہِے جو مَنڳݨاں کِین٘ویں ہِے۔\nمیݙی مَدَت کرو، میں سبزی خور نِمھیں پر کݙان٘ہیں کݙان٘ہیں میں چَنڳی صِحت آلا کھاݨاں کھاوݨ دی کوشِت کرین٘دا ہاں۔\nکیوں نِھیں، میکوں ہِک سِدّھا سنیہڑا گَھل ݙیؤ تاں جو اَساں رَل تے اِین٘دے بارے ڳالھ مُہاڑ کر سڳوں، ٹِھیک ہِے؟
ڈیٹا دی جَاچ دا مضمون سائنس دے شعبے آلے شاگرداں کِیتے ڈَھالیا ڳِیا ہِے۔ شاگرداں وِچ کمپیوٹنگ، شماریات، اَتے ڈیٹا دے تجزیے دے کورساں دے ذریعے ڈیٹا اُتے مبنی ڳِݨ مُن٘جھ دے مسئلے حَل کرݨ دا ڳُݨ پیدا تِھیسی۔ اِیں نصاب دا پَن٘دھ شاگرداں کوں پَرَکھ دے ذریعے اِنّھاں ڳُݨاں کوں وَرت تے ہِک خاص شعبے وِچ وَدھ سِکّھݨ دا موقع وی ݙیسی۔ ایہ مضمون حساب اُتے زور ݙین٘دا ہِے اَتے اُنّھاں شاگرداں کِیتے چَنڳا ہِے ڄہڑے حساب، کمپیوٹر سائنس، اَتے وکّھو وَکھ مسئلیاں کوں حَل کرݨ وِچ مُہان٘درے ہِن۔
تُساں حالی ہِک رجسٹرڈ اکاؤنٹ مُتُّوں ڳول کِیتی ہِے۔ ڄیکر تُہاݙا اَساݙے نال پہلاں اکاؤنٹ کائنی، تاں مہربانی کر تے اِتّھاں رجسٹر تِھیؤو۔ رجسٹریشن اَن٘جو اَن٘ج خاصیتاں دے ذریعے تُہاݙی ڳول کوں سَوکھا بݨیسی ڄہڑیاں صرف رجسٹرڈ ورتݨ آلیاں کِیتے موجود ہِن۔

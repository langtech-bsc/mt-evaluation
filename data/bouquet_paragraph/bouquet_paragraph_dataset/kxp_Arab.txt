دماغ نِي صحت هارُو هر روز ويلا اُوٺو هانَ پوتان نان ڳوۮڙان واۯو۔ اين چيڙ ورزش ڪرو ڪي هوار نا تازا وائيرا مان تمارُون ڏِيل هانَ دماغ ٻيئِي صحتمند ري۔ هانَ زي ٿئِي ڪي تو ۮُوڌ پِيئو هانَ ڪانڪ هڌرُون ڀڻو۔ گھڻان ڦَل کو ڪي تمين پوتان نين تندرست محسوس ڪرو هانَ تمارو ۮاڙو خُوشِي ٿِي شُروع ٿائي!
ديسِي مشالا واۯِي چانه ڪاڙهيا هارُو تمين نين ۮُوڌ، اَدرڪ، لاڪڙان، ڦوٽا، ڪاۯامرِي، کنڊ هانَ پاڻِي نِي ضرُورت پڙشي۔ ٻڌانئين ٿِي مورِ اَدرڪ نين ڪُوٽِينَ پاڻِي مان اُوڪاۯو۔ پسي ٻِيزا ٻڌائي مشالا اُوڪۯتا پاڻِي مان لاکو هانَ گھڙِيڪ هزِي اُوڪۯوا آلو۔ زار اُوڪۯتُون پاڻِي خُوشبُودار ٿئِي زائي تار اين مان هۯُوئي هۯُوئي اُوننُون ۮُوڌ نامو۔ زار چانه هڌري هڌري اُوڪۯِي ري، تار اين نين ساۯِينَ ڪوپ مان لئِينَ عيش ڪرو!
هانڀۯِ، شُون مُون آز بازار مانئين ٿِي تُون هارُو ڪانڪ لِيئاوون؟\nمنين ڪانئين نٿِي زوئيتُون، پَڻ تُون پاڻ هارُو ايڪ نوَي ساڙِي چَم نٿِي ليتِي؟\nمُون تو هميشان ساڙيُون ليتِي رُون سُون، پَڻ تمين پاڻ هارُو ڪانئين نٿِي ليتا۔\nپسي شُون خيال سي ڪي مُون ايڪ نوَوي شرٽ لِيئاوون، پَڻ ايوِي لاوون زي تُون نين هڌرِي لاڳي۔\nپِيۯِي شرٽ تمان ماٿي هڌرِي لاڳشي!\nتُون نين هميشان پِيۯو رنڳ هڌرو لاڳي سي!
سائِين، هوارنان منين اَڌ ۮاڙا نِي موڪل زويئي سي۔\nشُون ڇُون، ٻڌانئين ٺِيڪ تو سي؟\nمارِي ٻينڃ نين سُوڪرُون ڌاڻُون ڇُون سي، مُون اين نين زوئا زُون سُون۔\nواه، مُبارڪ هوئي!
آنند، آز تمان نين چيوون لاڳي رُون سي؟\nشُون تمين ڪالئي شوِيتا ڀيۯا ڊاڪٽر ڪنين جھا تا؟\nهووي نيها، مُون آنند نين ڪالئي ڊاڪٽر ڪنَ لئِينَ جھو تو۔\nلاڳي سي آنند اُونجھو پڙيو سي۔\nنمسڪار دوستو، مُون ٺِيڪ سُون۔ ڊاڪٽران نِي دوايانئي ڪمال ڪرِي لاڇو!
ما مارِي اسڪُول واۯان گھُموا زائي ران سي هانَ منين پَڻ اِيئان ڀيۯُون زاوون سي؟\nمنين ڪوئي اعتراض نٿِي پَڻ تُون نين تارٻا ٿِي پُوسوون پڙشي!\nنا، مارٻا تو نا پاڙشي پرو۔\nاِي تو تُون نين تار کٻير پڙشي زار تُون پُوسِيش!\nٺِيڪ سي مُون اِيئان ٿِي پُوسُون سُون۔
اوئي هوئي، اُونگھ مُون ٿِي رِيشي ٻۯِي پرِي سي!\nچُٽڪِي، تُون اَرڌِي راچ نا بي سُرا آواز مان گانُون چَم ڳاوي رِي سي؟\nاو ٻائِي، منين اُونگھ نٿِي آوَتِي۔\nزي تُون نين اُونگھ نٿِي آوَتِي تو اين نُون مطلب اِي نٿِي ڪي پُورِي اَڪائِي نِي اُونگھ حرام ڪري!\nاو ٻائِي، اُونگھ نُون راز تو هنڀۯائي۔\nٿوڙُونڪ اُونُون ۮُوڌ پِي هانَ آنڇُون بند ڪرِينَ اُونگھائِي زا!
فروگِيئي آز ڊاڪٽر نُون نمبر لِيڌُون سي۔ فروگِِي هانَ اين نِي ما ڊاڪٽر ڪنَ زائي ران سي۔ ڊاڪٽر نُون نام روهيت سي هانَ اِي فروگِي نو مِتر سي۔ فروگِي نِي ما اين نين ڪي زي ڪي ۮانت صاف ڪروا مت ڀُولزي هانَ نوِي چڊِي پيرزي چمڪي آز اين نين ڊاڪٽر نين وَتاڙوو سي۔ پَڻ فروگِي چڊِي پيروا ڀُولِي زائي سي تو اين نِي ما اين هارُو ڪاڳۯ نِي چڊِي ٺاهي سي۔ ڊاڪٽر روهيت فروگِي نين چيڪ ڪري سي هانَ ڪي سي فروگِي تُون هاوَ ٺِيڪ ٺاڪ سي۔
اِي هۯُوئي هۯُوئي ڍُوڪڙِي آوي سي هانَ اين نِي آنڇان مان ايڪ عجيب وات هوئي سي۔ ايم لاڳي ڪي ڪي هر وَک ماٿي وَکت اُوڀو رئِي زائي سي هانَ جَڳت نِي ٻڌِي خُوشِي اين نا هشا مان ري سي۔ مُون اين نين زوون سي تو ايم لاڳي سي ڪي اين نُون مَن مارِي ڌڙڪن ٿِي سانٽل سي۔ مُون ڪانئين ڪِيڌا ٿِي مورِ اين نِي آنڇان مان کووَرائِي زُون سُون، زانئين هر پل مان اَمين ٻيئِي هان سان۔
اينڻِيئي مُون هانمُون زويُون هانَ هانشِينَ ڪِيڌُون ڪي تُون ايوو لاڳي ري سي زانڻي گڌاڙي ٽچ موبائيل لِيڌُون هوئي۔ مين مُونڍُون ٻڳاڙيُون هانَ ڪِيڌُون ڪي، تو شُون تُون وَري موٽِي آداگرانڻِي لاڳي ري سي؟ پسي اَمين ٻيئِي هشِي هشِينَ پڙيان ٺان۔ پسي اينڻِيئي ڪِيڌُون ڪي زي ڪوئي گڌاڙو ٽچ موبائيل هاڪوا شِيکي پرو تو شُون تُون اين نين همُون ڪريا مان مارِي مدد ڪرِيش؟ مين مارُون هانهُون روچا نِي ڪوشش ڪرِي هانَ ڪِيڌُون ڪي، زوئي مُون گڌاڙو سُون هانَ تُون ٿِي وات ڪرِي ڪرِينَ ٿاڪتو نٿِي!
آز مُون ماران دوستان ڀيۯو ايڪ انڊو چائينِيز ريسٽورينٽ جھو تو۔ زار اَمين ريسٽورينٽ مانئين جھا، ٻڌانئين ٿِي مور اِيئانئي اَمارو هڌرو استقبال ڪريو هانَ اَمان نين پاڻِي آليُون۔ اِي ايڪ هڌرو اِشارو هتو زي هميشاه تمان نين ايڪ آرام واۯو احساس اَلاوي سي۔ ماحول پَڻ اَلاهِي سڪُونَ واۯو هتو، ڀيۯِي ايڪ جدِيد جھلڪ هتِي زي کاوا هارُو بلڪل مناسب هتِي۔ اَمين اَڃين اَلاهِي نمُونا نان کاوا آزمايان زي هاوَ انڊو چائينِيز ذائيقان نو هڌرو نمُونو هتو۔ سروس ٿوڙِي سست هتِي توئي اَمين اَلاهِي اَنجوئي ڪريو۔
مين لنڪڊن پروفائيل اپ ڊيٽ ڪرِي پرِي سي۔ هوي وَکت آيو رو سي ڪي تمين پوتان نا هنر، تجربا هانَ ڪاميابيان نين صحيح نمُوني پيش ڪرو۔ مُون هميشان ڀروسو راکُون سُون ڪي زار آپي پوتان ماٿي ڀروسو ڪران سان هانَ پوتان نا تجربا ٻِيزان نين هنڀۯاوان سان تو دنيا پَڻ تمارِي خود اعتمادِي نين محسُوس ڪري سي۔ پوتان نِي ترقِي ايڪ فنَ سي هانَ اِي تار فائيدا واۯِي ٿائي سي زار اين نين سهِي نمُوني ڪرِي زائي!
زي تمين پوتان نِي خوٻصُورتِي وَڌاوا ڪرو سو تو آ سُرخِي بلڪل تمان هارُو سي۔ آ سُرخِي نا رُڄو تمارا هوٺان نين نرم هانَ رنڳِيلا راکي سي، پَڻ اين نو رنڳ ايڪ خاص چمڪ پَڻ آلي سي زي تمارِي خُوبصُورتِي نين وَڌاوي سي۔ ڀلي تمين پارٽِي مان سادان لاڳوا ڪرو يان فيشن واۯان، آ سُرخِي هر نمُوني ٿِي تماران هوٺان نين هڌرا ڪري سي۔ آنئين نو اثر ڌير هُوڌِي ري سي۔ هانَ هووي، آنئين نِي هڌرِي پيڪنگ پَڻ اين نين ايڪ خاص تحفو ٺاهي سي۔ ايڪ هاس نو پسند نو تجربو۔ ايڪ چڪر آنئين نين آزماوِينَ زوئو، پسي تمين پاڻ محسوس ڪرشو ڪي آ تماران هوٺان نين چيوا هڌرا ڪري سي
آ شِيز ايٽلِي هڌرِي سي ڪي ايم لاڳي سي ڪي آ شِيز مان ڪوئي آسمانِي ۮيوِي موجود هوئي۔\nآ نظارو آپڻا منَ نين آرام آلي سي۔\nمارو آتما آنئين مان ري سي، آ چيوون هڌرُون ڦوٽُو سي۔
نمِيتائي انپم نِي پيشڪش ماٿي سهِي ڪِيڌُون ڪي اُو نوان مَنکان نو نا جائز فائيدو اُوپاڙي رو سي۔\nپيوش هانَ رِيتش ڳرِيٻ سي، عزت واۯا هانَ ايماندار سي هانَ اِي هميشان مَنکان نِي مدد هارُو آوي را سي۔\nانپم هاس نو اَلاهِي هُوشيار سي، پَڻ منين نٿِي کٻير ڪي اُو آ ۮاڙان مان شُون ڪري رو سي۔\nانپم آنڻان ۮاڙان مان اَلاهِي ازايان ڪمينٽ ڪري رو سي۔\nترينڻمو سودو: اُو هاس نِي ڀٽڪِي جھئِي تِي هانَ پيوش اين هارُو سهِي هتو۔
اڙي دوست، آ نظارو زوئِينَ ايم لاڳي سي ڪي زمائِي ڪام واۯو هانَ ايڪشن لِيڌا واۯو آۮمِي سي۔\nاينڻي ايٽلو اسٽائل واوڙيو زَم ڪوئي انڊين اداڪار مورِ ٿِي هانمو آوي هانَ پسي زاتو ري۔\nمنين اَلاهِي مزو آيو زار زمائِيئي ڪِيڌُون ڪي آئي نِي ديواران مان دريُون چَم نٿِي؟\nاَڙي دوست، زمائِي نو اِي نظارو چيوو هڌرو هتو۔
گھر مان خطرا نو خيال راکو هانَ پڙيا واۯِي شِيزان ٿِي سيٽي رو۔ گھر مان هانَ ٻارئي محفوظ جڳايُون زوئو هانَ زخمِي مَنکان نين اَڃين لئِي زايا مان مدد ڪرو۔ ايمرجنسِي نين دوران ريسڪيو سروس ٿِي رابطو ڪريا هارُو ايڪ منصُوبو ٺاهيا نِي ڪوشش ڪرو۔ برادرِي نِي مدد ڪرو هانَ اُو مسئلا ساوا ڪرو زين ماٿي توجه آلوِي زويئي۔
آنڻا دور مان ٽيڪنالوجِي نين محفوظ هانَ سهِي استعمال هارُو قانُون ضرُورِي سي۔ آيا واۯان ۮوهہ سالان مان انڊيا ٽيڪنالوجِي مان ايڪ طاقتور مُلڪ ٿئِي زاشي۔ ايوا وَکت مان ڊِيجِيٽل عوامِي نظام ايڪ سيڪنڊ نا زَم ڪام ڪرشي هانَ اين نين رُڪاوٽ نا همزيون زائي۔ انڊيا عالمِي ڀلائِي هارُو پوتان نِي ڊِيجِيٽل عوامِي بنياد جڳت نين آليا هارُو تيار سي۔ انڊيا انسانِي حقوق نا تحفظ هانَ عالمِي خُوشحالِي هارُو ترقي هانمي ڪام جارِي راکشي۔
روڊ مٽيا ٽاڻي زيبرا ڪراسنگ نو استعمال ڪرو۔\nپيٽرول پمپئي موبائيل استعمال ڪروون منع سي۔\nوسڪِي پِيڌا چيڙ ڳاڏِي هاڪوِي خطرناڪ سي۔\nپوتان نِي ڳاڏِي نين آر۔ ٽِي۔ او۔ مان داخلا لازمِي ڪراوو۔
آ سال ميۯا مان اَلاهِي رَش هتِي، خاص ڪرِينَ هانز نين ٽانڻي۔ دُڪان واۯا اَلاهِي خُوش هتا چمڪي شِيزُون اَلاهِي ويساتِي تِي۔ مَنک پينڳان، چاٽ هانَ جليٻيُون کائِينَ اَلاهِي رازِي ٿاتان تان۔ ميۯا مان رَش گھڻِي هتِي اين ٿِي ڪرِينَ پُولِيش پَڻ هتِي۔ رَش ٿِي ڪرِينَ شهر مان ڳاڏيُون پَڻ اُوڀِي ريتيُون تيُون۔
سوشل ميڊيائي وات سينچ ڪريا نو طرِيقو بدلِي لاڇو سي۔ هوَي مَنک ايڪ ٻِيزان ٿِي مورِ ٿِي ڪرتان جلدِي وات سينچ ڪرِي ڪي سي۔ ايڪ وات آ پَڻ هاسِي سي ڪي مَنکان مان پريشانِي هانَ غلط معلومات پکڙيا نو خطرو وَڌِي جھو سي۔ آ رُڄو مَنکان نا وِيچار نٿِي بڌلاوَتُون پَڻ معاشرا مان وَڍويڙ هانَ نفرت پَڻ پکيڙي رُون سي زين ٿِي ڪرِينَ ٻڌانئين نا مِلِينَ فيصلا ڪروا مَشڪل ٿاتُون زائي رُون سي۔
زي تمان نين سموسا زويئي سي تو اَمين سموسا آلشان هانَ زي تمان نين سموسا ٺاهيا نُون سامان زويئي سي تو اَمين اِي پَڻ تمان نين آلشان، چمڪي اَمين هر شِيز پوتاڙان سان۔\nتمين پوتان نا موبائيل نين گھڙِي گھڙِي نا زوئو چمڪي تمارُون سامان 10 منٽان مان پوتِي زاشي۔\nڪرڪيٽ ميچ سي هانَ گھرئي چپس نٿِي، اَمين پوتاڙشان، آ هاس سي۔

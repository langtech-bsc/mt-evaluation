རོ་ཁྱི་སྨོ་ བྱི་ལི? ངེའི་ཨ་ལུ་ཚུ་གིས་ ལོ་དག་པ་ཅིག་གི་རིང་ལུ་ རོ་ཁྱི་ཡང་ན་ བྱི་ལི་ཅིག་འབག་གནང་ཟེར་ཉན་ར་མ་བཏུབ། གནས་སྐབས་འདི་ལུ་མ་བཏུབ་ཅིག་མེད། ཨིན་རུང་ ང་གིས་འབད་བ་ཅིན་ དོན་ཚན་འདི་གི་ཐོག་ལུ་ ཞིབ་འཚོལ་མཐིལ་ཕྱྣིན་སྦེ་འབད་དགོཔ་བཟུམ་མཐོང་མས། རོ་ཁྱི་གསོ་འདི་ གནམ་མེད་ས་མེད་ལཱ་ཨིན། ཁོང་ཚུ་ ལེགས་ཤོམ་སྦེ་བདག་འཛིན་འབད་དགོ་པའི་ཁར་ གནམ་མེད་ས་མེད་ཟད་འགྲོ་འགྱོཝ་ཨིན། རོ་ཁྱི་དང་ཕྱདཔ་ད་ བྱི་ལི་གསོ་ནི་འདི་ འཇམ་ཏོང་ཏོ་ཅིག་ཨིན་རུང་ བྱི་ལི་འདི་ རོ་ཁྱི་དང་ཕྱདཔ་ད་ མཛའ་མཐུན་ཅན་མེདཔ་ཨིན། དེ་ཡང་ དཔྱེ་གཏམ་ལས་ “རྟ་གསོ་བའི་དྲིན་ལན་རྡོག་ཐེ་གིས་འཇལ།” ཟེར་དོ་བཟུམ་ བྱི་ལི་ཚུ་གིས་ དེ་སྦེ་འབད་ནི་ཡོད། ཇོ་བདག་ཚུ་ ཁྱིམ་ནང་དུས་ཡུན་རིངམོ་སྦེ་མ་སྡོདཔ་ད་ གཅེས་པའི་སེམས་ཅན་འདི་ ག་དེ་སྦེ་བཞག་ནི་ཨིན་ན་གི་ མནོ་བསམ་ཡང་གཏང་དགོ། རོ་ཁྱི་ཚུ་ ཁོང་ར་རྐྲངམ་གཅིག་བཞག་ནི་མི་འོང་།

ཁྱོད་ར་ཤང་ཧའི་ལུ་འགྱོ་ནིའི་འཆར་གཞི་ཡོད་པ་ཅིན་ ག་ཅི་ཟ་ནི་ཨིན་ན་ར་མ་ཤེས་པ་ཅིན་ འ་ནཱི་ལམ་སྟོན་ལུ་བལྟ། འདི་ནང་ ས་གནས་ཀྱི་མི་ཚུ་གིས་དགའ་མི་ ཡོངས་གྲགས་ཅན་གྱི་བཟའ་ཁང་ཚུ་གི་སྐོར་ལས་ བཀོད་དེ་ཡོདཔ་ལས་ ཁྱོད་ར་བློ་འགྱོད་བསྐྱེད་ར་མི་བསྐྱེད། ཤང་ཧའི་གི་ས་གནས་ཀྱི་ཡོངས་གྲགས་ཅན་གྱི་བཟའ་ཁང་ཚུ་གི་སྐོར་ལས་ མི་ཅི་ལེན་གྱིས་ ལོ་ལྔ་གི་རིང་མ་ཆད་པར་ རྒྱབ་སྣོན་འབད་དེ་ཡོདཔ་ཨིན། ལྟོ་ཟ་ས་འདི་ཆུང་ཀུ་ཅིག་ལས་བརྒལ་མེད་རུང་ འདི་ནང་ལུ་ག་ཅི་དགོཔ་ཅིག་ཡོདཔ་ཨིན། བཟའ་ཁང་གི་ས་སྒོ་འདི་སྒོ་ཡངས་དྲགས་སྦེ་ཡོད་པའི་ཁར་ གོང་ཚད་ཁེ་ཏོག་ཏོ་སྦེ་ཡོདཔ་ཨིན། དེ་འབདཝ་ལས་ དགའ་འཛོམས་ནང་འགྱོ་དགོ་རུང་ འཛོམས་འདུ་ཅིག་འབད་དགོ་རུང་ ས་སྒོ་འདི་དྲག་ཤོས་ཅིག་ཨིན། བཟའ་ཁང་འདི་ཚུ་ནང་འགྱོ་བའི་སྐབས་ ཉིནམ་གཅིག་གི་ཧེ་མ་ལས་སླབ་བཞག་དགོཔ་ཨིན། དེ་མེན་ ས་སྒོ་འདི་ཁར་ལྷོདཔ་ད་ བསྒུག་སྡོད་དགོཔ་འཐོན་འོང་། མི་རེ་ལུ་ཆ་སྙོམས་གོང་ཚད་ ¥༡༠༠ ཧོང་ཧིང་བཟའ་ཁང་: ཨང་ ༢། བར་ལམ་ ༧༡༦ ། ཕུ་ཟིང་ཞོང་ལུ།
ངེ་གི་ཏའི་ཝན་གྱིས་ཆ་རོགས་ལུ་དྲི་བལྟཝ་ད་ ཁོང་རའི་ཟིན་བྲིས་ཚུ་ གནས་སྟངས་ཀྱི་སྟབས་བདེ་ཧིང་དང་འཁྲིལ་ སྔོན་མའི་རྒྱ་ནག་གི་ཡི་གུ་དང་དེང་སང་གི་ཡི་གུ་གཉིས་ཆ་ར་ནང་བྲིཝ་ཨིན་ཟེར་སླབ་དེས།\nང་གིས་ རྒྱ་ནག་གི་སྔར་སྲོལ་ཡི་གུ་འདི་ལྷབ་ར་མ་ལྷབ། ང་གིས་ འ་ནཱི་ཡི་གུ་ཚུ་རེ་རེ་སྦེ་ལྷག་ནི་འགོ་བཙུགས་ཅི། གལ་སྲིད་ ཡི་གུ་ཚུ་ལྷག་མ་ཤེས་པ་ཅིན་ ཕོག་ཚད་གུར་དཔག་སྟེ་སླབ་ནི་ཡོད། དེ་འབདཝ་ད་ ཡུདཔ་ཅིག་ལས་ ང་གིས་སྟབས་མ་བདེཝ་མེད་པར་ ལེགས་ཤོམ་སྦེ་ར་ལྷབ་ཚུགསཔ་ཅི།\nགོ་ལེ་སྦེ་ལྷག་སྟེ་ ཁྱོད་ཀྱིས་ཡི་གུ་ཚུ་ཤེས་ཚུགས་རུང་ འ་ནཱི་འདི་འུ་སྡུག་སི་སི་ཅིག་ཡོད། ང་གིས་ རྒྱ་ནག་གི་སྔོན་མའི་ཡི་གུ་ཚུ་ལྷག་ནི་འདི་དགའ་ར་མི་དགའ་བས།\nཁྱོད་ཀྱིས་རྒྱ་ནག་གི་སྔོན་མའི་ཡི་གུ་ཚུ་མ་ཤེས་པ་ཅིན་ དང་ཕུ་གི་དཔེ་ཆ་མང་ཤོས་ར་ཧ་གོ་མི་ཚུགས།\nང་མཁས་ཆོག་མེན། ཨིན་རུང་ མི་ཕལ་པའི་ལོགས་ལས་འབད་བ་ཅིན་ གལ་སྲིད་ཁྱོད་ཀྱིས་མ་ཤེས་པ་ཅིན་ མ་པ་ལས་མི་ཤེས། འ་ནཱི་འདི་ང་ལུ་འབད་བ་ཅིན་ ཁྱད་ར་མེད། དེ་ཡང་ ག་ར་འབད་རུང་ དང་ཕུའི་དཔེ་ཆ་བལྟ་ནི་ལུ་ མཁས་པ་མ་འགྱོ་རུང་བཏུབ།

རེ་ལི་གི་ཤོག་འཛིན་ཉོ་བའི་སྐབས་ ངེའི་མི་ཁུངས་ངོ་སྤྲོད་ལག་ཁྱེར་འདི་ ག་ཅི་སྦེ་སྟོན་དགོཔ་སྨོ?\nརེ་ལི་གི་ཞབས་ཏོག་ག་བཟུམ་ཨིན་རུང་ ཤོག་འཛིན་ཉོ་བའི་སྐབས་ལུ་ འགྲོ་འགྲུལ་པ་རང་སོའི་ མིང་དང་ངོ་སྤྲོད་ལག་ཁྱེར་ཨང་ཚུ་ ངོ་མ་ལག་ལེན་འཐབ་དགོཔ་སྦེ་བཟོ་སྟེ་འདུག། འབྲེལ་བ་ཡོད་པའི་ རང་སོའི་མིང་དང་ངོ་སྤྲོད་ལག་ཁྱེར་ངོ་མ་ལག་ལེན་འཐབ་དགོ་པའི་ འཛིན་སྐྱོང་ཐབས་ལམ་ལུགས་ སྐྱེལ་༼འདྲེན་ལྷན་ཁག་ ༢༠༠༢ བཀའ་རྒྱ་ཨང་༣༩༽ དང་འཁྲིལཝ་ད་ ཤོག་འཛིན་ཚུ་ཉོ་བའི་སྐབས་ལུ་ རང་སོའི་ཆ་གནས་ཡོད་པའི་ཡིག་ཆ་ཚུ་ལག་ལེན་འཐབ་དགོཔ་ཨིན། ཆ་གནས་ཡོད་པའི་ཡིག་ཆ: རྒྱ་ནག་གཞུང་གི་མི་ཁུངས་ངོ་སྤྲོད་ལག་ཁྱེར། ཧོང་ཀོང་ མ་ཀའུ་དང་ཏཱའི་ཝན་ལུ་སྡོད་མི་ཚུ་ལུ་ཨིན་པ་ཅིན་ སྡོད་ཆོག་པའི་ཆོག་ཐམ་འདི་དགོ། དེ་ལས་ ཕྱིའི་ཆོག་ཐམ་ཚུ་དགོ།

ཨམ་སྲུ་སྐྱེས་ལོ་ ༣༠ ཡན་ཆད་འབད་མི་ཚུ་ལུ་ ཕྱི་རུའི་སྨན་དྲག་ཤོས།\nགདོང་གུར་དབུར་ནིའི་སྨན་ལེ་ཤ་འདུག། ཁྱོད་ཀྱིས་ག་འདི་གདམ་ཁ་རྐྱབ་ནི་ཨིན་ན་ ཤེས་ག? ང་གིས་སླབ་གེ་ གདོང་བདག་འཛིན་འབད་ནི་ཟེར་མི་འདི་ སྨན་ལེ་ཤ་དབུར་ནི་འདི་མེན། དེ་འབདཝ་ད་ གཟུགས་ཁར་ཕན་པའི་སྨན་ལེགས་ཤོམ་ གདམ་ཁ་རྐྱབ་ནི་འདི་ཨིན། ཕྱི་རུ་དབུར་ནིའི་སྨན་འདི་ དམིགས་བསལ་གྱི་ ཨམ་སྲུ་སྐྱེས་ལོ་ ༣༠ ཡན་ཆད་འབད་མི་ཚུ་གི་དོན་ལུ་བཟོ་བཟོཝ་ཨིན། དེ་གིས་ གཞོནམ་བཟུམ་སྦེ་མཐོང་ཚུགསཔ་བཟོ་བའི་ཁར་ གཟུགས་ཁར་གཉེར་མ་མེད་པར་དཀརཔོ་སྦེ་བཞག་ཚུགས་པའི་ཁར་ ནག་ཐིག་ཡོད་མི་ཚུ་ཡང་མེདཔ་བཟོཝ་ཨིན། ང་བཅས་ཀྱི་སྨན་གྱིས་ འ་ནཱི་དཀའ་ངལ་ག་ར་སེལ་ཚུགས། ཁྱོད་ར་ཞལ་མ་ཉལ་བའི་སྔོན་མ་ གདོང་འཁྱུ་ཞིནམ་ལས་ དུམ་གྲ་ཅིག་དབུར་དགོཔ་ཨིན། ཉིནམ་དེའི་ནངས་པ་ར་ ཁྱོད་རའི་གདོང་ག་དེམ་ཅིག་ལེགས་ཤོམ་འགྱོ་ནུག་ག་མཐོང་ཚུགས།\n⬇️⬇️⬇️རིན་མེད་སྟོང་པ་སྦེ་དཔེ་ཅིག་བལྟ་ནིའི་དོན་ལུ་ འོག་ལུ་ཡོད་པའི་འབྲེལ་མཐུད་གུར་ཨེབ་གནང་⬇️⬇️⬇️

ང་བཅས་ཀྱི་སྐོར་ལས།\nང་བཅས་ མང་ཤོས་ར་ ཚོང་འབྲེལ་ཆེ་བ་གིས་ འབྲེལ་བ་འཐབ་ཨིན་ཟེར་མི་འདི་བདེན་པ་ཨིན་རུང་ ང་བཅས་ཀྱིས་ ཚོང་འབྲེལ་ཆེ་ཆུང་ག་བཟུམ་ཅི་ལུ་ཡང་ ཞབས་ཏོག་འདི་བྱིནམ་ཨིན། ང་བཅས་ཀྱི་ལཱ་འདི་ ཁྱོད་ར་ལུ་ ཕན་ཐབས་འབད་ནི་འདི་ཨིན། ང་བཅས་ཀྱི་དམིགས་པ་འདི་ ཁྱོད་ ལས་བྱེདཔ་གིས་ ཁོང་རའི་ བརྩོན་ཤུགས་ཧྲིལ་བུམ་བསྐྱེད་བཅུག་སྟེ་ ཁྱོད་རའི་དམིགས་པ་གྲུབ་ཐབས་ལུ་ བདག་འཛིན་པའི་རིམ་པ་དྲག་ཤོས་ཕུལ་ནི་འདི་ཨིན། ང་བཅས་རའི་མདུན་ལམ་འདི་ ལོ་ངོ་ ༡༠ གི་ཧེ་མ་ ལཱ་དང་མི་ཚེ་འདྲ་མཉམ་མེད་པའི་ དཀའ་བསྔལ་གྱི་གནོད་མི་ཚུ་ལུ་ ཐབས་ལམ་ཕུལ་ནི་གི ཡིད་ཆེས་བརྟན་པོ་ཐོག་ལས་འགོ་བཙུགས་ཡོདཔ་ཨིན། ང་བཅས་ཀྱི་ ཞབས་ཏོག་དང་ ཐབས་ལམ་ག་ར་ ཞིབ་དཔྱད་འབད་གནང་།

མོ་གིས་ང་ལུ་བལྟ་སྟེ་བགའ་སྦེ་ དེ་སྦེ་སླབ་ཅི་ ‘ཁྱོད་བོང་ཀུ་གིས་འགྲུལ་འཕྲིན་ཉོ་ཉོཝ་བཟུམ་མཐོངམ་མས།’ ང་གིས་གདོང་གི་རྣམ་འགྱུར་བཟོ་སྦེ་སླབ་ཅི་ ‘སྟེ་ ག་ཅི་ ཁྱོད་ཀྱིས་ཁྱད་འཕགས་ཅན་གྱི་དཔའ་མོ་བཟུམ་མཐོངམ་མས་ག?’ དེ་ལས་ ང་བཅས་གཉིས་ཆ་ར་ཚར་གཅིག་གུ་བགའ་ཡི། དེ་ལས་ མོ་གིས་ ‘གལ་སྲིད་ བོང་ཀུ་གིས་འགྲུལ་འཕྲིན་ལག་ལེན་འཐབ་ཚུགས་པ་ཅིན་ ཁྱོད་ཀྱིས་ང་ལུ་འདི་བཅོ་ཁ་རྐྱབ་ནི་ལུ་ ཆ་རོགས་འབད་འོང་ག?’ ཟེར་སླབ་ཅི། ང་གིས་བགའ་ནི་འདི་མར་ཕབ་འབད་ནི་བརྩི་རྐྱབ་ཏེ་ དེ་སྦེ་སླབ་ཅི་ ‘བལྟ་ཤིག ང་བོང་ཀུ་ཨིནམ་ལས་ ཁྱོད་དང་གཅིག་ཁར་བློ་སླབ་ད་ལུ་ ནམ་ཡང་འུ་སྡུག་ནི་མེད་’ ཟེར་སླབ་ཅི།

གལ་སྲིད་ཁྱོད་རའི་ངོ་རིས་ཡར་རྒྱས་གཏང་དགོ་པ་ཅིན་ ཨ་ནཱི་ཁ་ཚོན་གྱི་འོད་རྩི་འདི་ཁྱོད་ཀྱི་དོན་ལུ་ཨིན། ཨ་ནཱི་ཁ་ཚོན་གྱི་འོད་རྩི་འདི་གིས་ ཁྱོད་རའི་ ཁ་མཆུ་འཇམ་ཅུག་ཅུ་དང་ཆུ་ལྡན་སྦེ་བཞགཔ་མ་ཚད་ འདི་གི་ཚོས་གཞི་གིས་ཡང་ དམིགས་བསལ་གྱི་འོད་ཆེམ་ཆེ་བཟོ་སྟེ་ ཁྱོད་རའི་མཛེས་ཆ་ཡར་དྲག་བཏངམ་ཨིན། ཁྱོད་ར་ ཉམས་ཆུང་སྦེ་འཇའ་ཆི་ཆི་སྦེ་མཐོང་དགོ་རུང་ ཡང་ན་ སྤྲོ་སྟོན་ཅིག་ནང་ གནམ་མེད་ས་མེད་མ་འདྲཝ་སྦེ་འཇའ་ཆི་ཆི་སྦེ་མཐོང་དགོ་རུང་ ཨ་ནཱི་ཁ་ཚོན་འོད་རྩི་གིས་ དུས་སྟོན་ག་ཅི་བཟུམ་གྱི་དོན་ལུ་འབད་རུང་ ཁྱོད་རའི་ཁ་མཆུ་ལེགས་ཤོམ་བཟོཝ་ཨིན། འདི་གིས་ དུས་ཡུན་རིངམོ་སྦེ་འགོརཝ་ཨིན། དེ་ལས་ རིན་ཐང་ཅན་གྱི་བཙུགས་སྣོད་འདི་གིས་ཡང་ དམིགས་བསལ་གྱི་ཁྱད་པར་ཅན་བཟོ་བའི་ཁར་ ལོངས་སྤྱོད་ལྡན་པའི་ཚོར་སྣང་ཡང་བཟོཝ་ཨིན། འདི་ སྔོན་བརྟག་ལག་ལེན་འཐབ་བལྟ་སྟེ་ དེ་གིས་ཁྱོད་རའི་ ཁ་མཆུ་ག་དེམ་ཅིག་ལེགས་ཤོམ་བཟོཝ་མས་ག་བལྟ།

ན་མི་ཏ་གིས་ ཨ་ནུ་པཱམ་གྱི་གྲོས་འཆར་ལུ་ ཁོ་གིས་ མི་གསརཔ་ཚུ་ “ཉེས་བཀོལ་” འབདཝ་མས་ཟེར་ བདེན་པ་སྦེ་སླབ་ཅི།\nཔི་ཡུཤ་དང་རི་ཏཤ་འདི་ མཉམ་ཆུང་ གུས་ཞབས་ཅན་དང་ དྲང་པོ་ཡོདཔ་དང་ ཁོང་ཚུ་ཨ་རྟག་ར་མི་ཚུ་ལུ་ཆ་རོགས་འབད་ནིའི་དོན་ལུ་ གྲ་སྒྲིག་འབད་ཡོདཔ་ཨིན།\nཨ་ནུ་པམ་འདི་ གནམ་མེད་ས་མེད་ཡོན་ཏན་ཅན་ཨིན་རུང་ ང་གིས་ ཁོ་ད་རིས་ཀྱི་དུས་ལུ་ག་ཅི་འབད་ནི་དཔའ་བཅམ་ཨིན་ན་མི་ཤེས་པས།\nཨ་ནུ་པམ་གྱིས་ད་རིས་ཀྱི་དུས་ལུ་ འབྱུང་ཁུངས་མེད་པའི་བསམ་འཆར་ཚུ་བཀོདཔ་མས།\nསྒྲིག་ཆ་གསུམ་པ། མོ་འདི་ཐད་རི་བ་རི་ཕྱི་འགྱུར་སྦེ་འཁྱིད་ཡོདཔ་དང་ པི་ཡུཤ་འདི་གདམ་ཁ་དྲག་ཤོས་ཅིག་ཨིནམ་འོང་།

མིང་གཏམ་ལུ་ཐོབ་དབང་དང་རང་གི་པར་གྱི་ཐོབ་དབང་འདི་ ཨིས་པེན་ལུ་ སྲིད་ཁྲིམས་དང་ འཛམ་གླིང་ཡོངས་ཀྱི་འགྲོ་བ་མིའི་དབང་ཆའི་གསལ་བསྒྲགས་ གཉིས་ཆ་ར་གིས་ ཉེན་སྲུང་འབད་དེ་ཡོད། དེ་འབདཝ་ལས་ རང་དབང་གི་བསམ་བརྗོད་འདི་ སྐབས་ཐོབ་ཀྱི་གནད་དོན་འདི་ནང་ཧ་གོ་དགོཔ་ཁག་ཆེཝ་ཨིན། དེ་ཡང་ ཁྲིམས་ལུགས་འདི་གིས་ མི་སེར་ཚུ་ དམའ་ཕབ་དང་ཁ་གཏང་ནི་ལས་ ཉེན་སྲུང་འབདཝ་ལས་ ཁོང་གི་མིང་གཏམ་ལུ་གནོད་ཉེན་འབྱུང་ནི་ཨིནམ་ལས་ཨིན། འདི་ཚུ་འབྱུང་པ་ཅིན་ མི་དེ་གིས་དེ་བཟུམ་སླབ་མི་འདི་ འགོ་ཐོག་དང་པ་ཁྲིམས་འགལ་གྱི་ལཱ་ཅིག་སྦེ་བརྩི་འོང་། ཨིན་ཏེ་ར་འབད་རུང་ འ་ནཱི་འདི་ གནམ་མེད་ས་མེད་ཁག་ཆེ། གལ་སྲིད་ ཨ་ནཱི་བསམ་འཆར་ཚུ་ ཡོངས་འབྲེལ་ཐོག་ལས་བཀོད་ཡོད་པ་ཅིན་ གལ་སྲིད་བསམ་འཆར་བཀོད་མི་གི་རྩོམ་པ་པོ་འདི་ ངོས་འཛིན་འབད་མ་ཚུགས་པ་ཅིན་ ཡོངས་འབྲེལ་འཆར་སྒོའི་བདག་སྐྱོང་པ་ལུ་ཡང་ ཁྲིམས་ལྡན་གྱི་ཕོག་ལམ་ཕོག་འོང་། དེ་མ་ཚད་ མི་སྡེའི་བརྡ་བརྒྱུད་ཚུ་ནང་ལུ་ དམའ་ཕབ་དང་སྐྱོན་བརྗོད་ཚུ་རྐྱབ་པའི་སྐབས་ ཉེས་བྱ་དང་བཀའ་གྲོལ་ཚུ་ བཀལ་ཏེ་ཡོད་མི་འདི་ཡང་ གྱངས་ཁ་ཆོདཔ་ལས་བརྟེན་ མང་སུ་ཅིག་སྦེ་ར་ ཁྱབ་སྤེལ་འགྱོ་དོ་ཡོདཔ་ཨིན་མས། དེ་འབདཝ་ལས་ ཁྱོད་རའི་ཡོངས་འབྲེལ་འཆར་སྒོ་ཡང་ན་བྲི་སྒོ་གི་བསམ་བཀོད་དང་གཅིག་ཁར་གཟབ་གཟབ་འབད།

སྤྱི་ཟླ་གསུམ་པའི་ནང་ ཆུ་རུད་ཀྱིས་ གནོད་པ་རྐྱབ་མི་ཚུ་ལུ་ གྲོགས་རམ་ཚུ་ ཁྱབ་བསྒྲགས་འབད་ནུག། གྲོགས་རམ་ཐུམ་ཚན་འདི་ ཉེར་མཁོ་ཆེ་བའི་ཅ་ཆས་འབྱང་མི་དང་འབྲེལ་བའི་ དཔལ་འབྱོར་གྱི་རིན་ངོ་ ཡར་སེང་རྐྱབ་ནིའི་དམིགས་གཏད་ཨིན། ཞུ་ཡིག་ཚུ་ སྤྱི་ཟླ་བཞི་པའི་སྤྱི་ཚེས་ ༦ ལས་ ༣༠ ཚུན་ གློག་རིག་ཐོག་ལས་བཙུགས་དགོཔ་དང་ དེ་གཉིས་ཆ་ར་ཚུད་དགོ། ཞུ་ཡིག་བཙུགས་ནིའི་ནང་འབོ་མི་ནང་ཚུད་ཡོད་པའི་ཁྲོམ་སྡེ་ཚུ་གི་ཐོ་འདི་ཟུར་སྦྲགས་དང་པ་ནང་མཐོང་ཚུགས།
